cmake_minimum_required(VERSION 3.24)
project(hotbox LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HOTBOX_BUILD_HARNESS "Build the GLFW and Dawn harness" ON)
option(HOTBOX_BUILD_TESTS "Build doctest tests" ON)
option(HOTBOX_WITH_SPDLOG "Enable spdlog in the library" ON)
option(HOTBOX_WITH_FREETYPE "Use the FreeType font rasterizer" ON)
option(HOTBOX_EMBED_FONTS "Embed static Inter faces and Lucide" ON)
set(HOTBOX_IMGUI_TARGET "" CACHE STRING "Existing Dear ImGui 1.92+ CMake target")
set(HOTBOX_DAWN_PREBUILT_DIR "" CACHE PATH "Install prefix containing a compatible Dawn CMake package")
set(HOTBOX_SANITIZER "" CACHE STRING "Instrument authored code with address or undefined sanitizer")
set_property(CACHE HOTBOX_SANITIZER PROPERTY STRINGS "" address undefined)

function(hotbox_require_vendor path)
    if(NOT EXISTS "${PROJECT_SOURCE_DIR}/vendor/${path}")
        message(FATAL_ERROR "Missing vendor/${path}. Run bash scripts/vendor.sh.")
    endif()
endfunction()

function(hotbox_instrument target)
    if(HOTBOX_SANITIZER)
        if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" OR MSVC)
            message(FATAL_ERROR "HOTBOX_SANITIZER requires GCC or Clang with the GNU driver")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=${HOTBOX_SANITIZER} -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=${HOTBOX_SANITIZER})
        if(HOTBOX_SANITIZER STREQUAL "undefined")
            target_compile_options(${target} PRIVATE -fno-sanitize-recover=undefined)
        endif()
    endif()
endfunction()

hotbox_require_vendor(imanim/im_anim.cpp)
hotbox_require_vendor(lunasvg/CMakeLists.txt)
hotbox_require_vendor(lucide/codepoints.hpp)
if(HOTBOX_IMGUI_TARGET)
    if(NOT TARGET "${HOTBOX_IMGUI_TARGET}")
        message(FATAL_ERROR "HOTBOX_IMGUI_TARGET must name an existing CMake target")
    endif()
    set(hotbox_imgui "${HOTBOX_IMGUI_TARGET}")
else()
    hotbox_require_vendor(imgui/imgui.cpp)
    add_library(hotbox_imgui STATIC
        vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp
        vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp)
    target_include_directories(hotbox_imgui SYSTEM PUBLIC vendor/imgui)
    set(hotbox_imgui hotbox_imgui)
endif()

add_library(hotbox_imanim STATIC vendor/imanim/im_anim.cpp)
target_include_directories(hotbox_imanim SYSTEM PUBLIC vendor/imanim)
target_link_libraries(hotbox_imanim PUBLIC ${hotbox_imgui})

set(LUNASVG_BUILD_EXAMPLES OFF)
set(LUNASVG_DISABLE_LOAD_SYSTEM_FONTS ON)
set(LUNASVG_DISABLE_EXTERNAL_RESOURCES ON)
set(USE_SYSTEM_PLUTOVG OFF)
if(NOT TARGET lunasvg::lunasvg)
    add_subdirectory(vendor/lunasvg EXCLUDE_FROM_ALL)
endif()

if((HOTBOX_WITH_SPDLOG OR HOTBOX_BUILD_HARNESS) AND NOT TARGET spdlog::spdlog)
    hotbox_require_vendor(spdlog/CMakeLists.txt)
    set(SPDLOG_BUILD_EXAMPLE OFF)
    set(SPDLOG_BUILD_TESTS OFF)
    set(SPDLOG_BUILD_BENCH OFF)
    add_subdirectory(vendor/spdlog EXCLUDE_FROM_ALL)
endif()

add_library(hotbox STATIC src/theme.cpp src/fonts.cpp src/log.cpp
    src/icons.cpp src/registry.cpp src/layout.cpp src/geometry.cpp src/gesture.cpp src/state.cpp
    src/atlas.cpp src/paint.cpp src/overlay.cpp src/motion.cpp src/hotbox.cpp)
add_library(hotbox::hotbox ALIAS hotbox)
target_include_directories(hotbox PUBLIC "${PROJECT_SOURCE_DIR}/include"
    PRIVATE "${PROJECT_SOURCE_DIR}/vendor/lucide")
target_compile_features(hotbox PUBLIC cxx_std_20)
target_link_libraries(hotbox PUBLIC ${hotbox_imgui} PRIVATE hotbox_imanim lunasvg::lunasvg)
if(APPLE)
    enable_language(OBJCXX)
    target_sources(hotbox PRIVATE src/motion_mac.mm src/system_appearance_mac.mm)
    target_link_libraries(hotbox PRIVATE "-framework AppKit")
elseif(WIN32)
    target_sources(hotbox PRIVATE src/system_appearance_win.cpp)
    target_link_libraries(hotbox PRIVATE user32 advapi32)
else()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(HOTBOX_GIO QUIET IMPORTED_TARGET gio-2.0)
        endif()
    endif()
    if(TARGET PkgConfig::HOTBOX_GIO)
        target_sources(hotbox PRIVATE src/system_appearance_linux.cpp)
        target_link_libraries(hotbox PRIVATE PkgConfig::HOTBOX_GIO)
    else()
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            message(STATUS "gio-2.0 unavailable. System appearance will default to Light.")
        endif()
        target_sources(hotbox PRIVATE src/system_appearance_fallback.cpp)
    endif()
endif()
if(MSVC)
    target_compile_options(hotbox PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(hotbox PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
hotbox_instrument(hotbox)

if(HOTBOX_WITH_SPDLOG)
    target_link_libraries(hotbox PRIVATE spdlog::spdlog)
    target_compile_definitions(hotbox PRIVATE HOTBOX_WITH_SPDLOG
        SPDLOG_ACTIVE_LEVEL=$<IF:$<CONFIG:Debug>,SPDLOG_LEVEL_DEBUG,SPDLOG_LEVEL_INFO>)
endif()

if(HOTBOX_WITH_FREETYPE)
    hotbox_require_vendor(imgui/misc/freetype/imgui_freetype.cpp)
    if(NOT TARGET freetype)
        hotbox_require_vendor(freetype/CMakeLists.txt)
        foreach(dep IN ITEMS ZLIB BZIP2 PNG HARFBUZZ BROTLI)
            set(FT_DISABLE_${dep} ON CACHE BOOL "Build the pinned FreeType without optional dependencies")
        endforeach()
        add_subdirectory(vendor/freetype EXCLUDE_FROM_ALL)
    endif()
    # Select this loader on the atlas. This also works with a host's stb-built ImGui.
    add_library(hotbox_freetype STATIC vendor/imgui/misc/freetype/imgui_freetype.cpp)
    target_include_directories(hotbox_freetype SYSTEM PUBLIC vendor/imgui/misc/freetype)
    target_link_libraries(hotbox_freetype PRIVATE freetype ${hotbox_imgui})
    target_link_libraries(hotbox PRIVATE hotbox_freetype)
    target_compile_definitions(hotbox PRIVATE HOTBOX_WITH_FREETYPE)
endif()

if(HOTBOX_EMBED_FONTS)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    hotbox_require_vendor(imgui/misc/fonts/binary_to_compressed_c.cpp)
    add_executable(hotbox_font_compressor vendor/imgui/misc/fonts/binary_to_compressed_c.cpp)
    set(font_header "${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_fonts.inc")
    set(font_files vendor/inter/Inter-Regular.ttf vendor/inter/Inter-Medium.ttf
        vendor/inter/Inter-SemiBold.ttf vendor/lucide/lucide.ttf)
    foreach(font IN LISTS font_files)
        if(NOT EXISTS "${PROJECT_SOURCE_DIR}/${font}")
            message(FATAL_ERROR "Missing ${font}. Run bash scripts/vendor.sh.")
        endif()
    endforeach()
    add_custom_command(OUTPUT "${font_header}"
        COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/embed_fonts.py"
            "$<TARGET_FILE:hotbox_font_compressor>" "${font_header}" ${font_files}
        DEPENDS hotbox_font_compressor scripts/embed_fonts.py ${font_files}
        WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" VERBATIM)
    target_sources(hotbox PRIVATE "${font_header}")
    target_include_directories(hotbox PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(hotbox PRIVATE HOTBOX_EMBED_FONTS)
endif()

if(HOTBOX_BUILD_HARNESS)
    include(cmake/Harness.cmake)
endif()

if(HOTBOX_BUILD_TESTS)
    hotbox_require_vendor(doctest/doctest/doctest.h)
    enable_testing()
    add_executable(hotbox_tests tests/bootstrap.cpp tests/core.cpp tests/interaction.cpp tests/paint.cpp)
    target_include_directories(hotbox_tests PRIVATE src)
    target_include_directories(hotbox_tests SYSTEM PRIVATE vendor/doctest vendor/lucide)
    target_link_libraries(hotbox_tests PRIVATE hotbox::hotbox)
    target_compile_definitions(hotbox_tests PRIVATE
        HOTBOX_TEST_FONT_DIR="${PROJECT_SOURCE_DIR}/vendor")
    if(HOTBOX_EMBED_FONTS)
        target_compile_definitions(hotbox_tests PRIVATE HOTBOX_TEST_EMBEDDED_FONTS)
    endif()
    hotbox_instrument(hotbox_tests)
    add_test(NAME hotbox_tests COMMAND hotbox_tests)
endif()
